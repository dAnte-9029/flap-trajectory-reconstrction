cmake_minimum_required(VERSION 3.24)
project(stage02_6 LANGUAGES C CXX)
find_package(Eigen3 REQUIRED NO_MODULE)
find_package(Ceres 2.2 REQUIRED)
find_package(Sophus REQUIRED)
set(STAGE01_FIX_PSI_CUTOFF ON CACHE BOOL "Audited PSI fix" FORCE)
add_subdirectory(../stage01 stage01)
get_target_property(_inc stage01_probe INCLUDE_DIRECTORIES)
get_target_property(_opts stage01_probe COMPILE_OPTIONS)
foreach(_target joint synthetic)
 add_executable(stage02_6_${_target} ${_target}.cpp "${GPTR_ROOT}/gptr/include/SE3JQ.cpp")
 target_compile_features(stage02_6_${_target} PRIVATE cxx_std_17)
 target_include_directories(stage02_6_${_target} PRIVATE ${_inc} ../stage01 ../stage02 ../stage02_3B)
 target_compile_options(stage02_6_${_target} PRIVATE ${_opts})
 target_link_libraries(stage02_6_${_target} PRIVATE Eigen3::Eigen Ceres::ceres Sophus::Sophus)
endforeach()
