cmake_minimum_required(VERSION 3.24)
project(stage01_gptr_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GPTR_ROOT "/home/zn/gptr" CACHE PATH "Read-only pinned GPTR checkout")
execute_process(COMMAND git -C "${GPTR_ROOT}" rev-parse HEAD OUTPUT_VARIABLE GPTR_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT GPTR_REV STREQUAL "2c0285a4f3141b131cc7b3aa77b3c6d979c0a2c2")
  message(FATAL_ERROR "This probe requires the audited GPTR revision")
endif()
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(Ceres 2.2 REQUIRED)
find_package(Sophus REQUIRED)
option(STAGE01_FIX_PSI_CUTOFF "Apply the recorded one-line PSI endpoint fix to a build-local header copy" OFF)
add_executable(stage01_probe probe.cpp "${GPTR_ROOT}/gptr/include/SE3JQ.cpp")
if(STAGE01_FIX_PSI_CUTOFF)
  set(_core "${GPTR_ROOT}/gptr/include/GaussianProcess.hpp")
  file(SHA256 "${_core}" _core_hash)
  if(NOT _core_hash STREQUAL "c87455374511b707aa30741efd39c4baee8e4505ecbf346a8cc5efb86b737680")
    message(FATAL_ERROR "PSI patch only reviewed for the exact audited header")
  endif()
  file(READ "${_core}" _core_text)
  # Keep the exact endpoint shortcut; do not erase the right endpoint's
  # influence over the first millisecond of every knot interval.
  string(REPLACE "if (Dtau < DOUBLE_EPSILON)" "if (Dtau == 0.0)" _core_patched "${_core_text}")
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/upstream_overlay")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/upstream_overlay/GaussianProcess.hpp" "${_core_patched}")
  target_include_directories(stage01_probe BEFORE PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/upstream_overlay")
  target_compile_definitions(stage01_probe PRIVATE STAGE01_FIX_PSI_CUTOFF=1)
endif()
target_include_directories(stage01_probe PRIVATE "${GPTR_ROOT}/gptr/include")
target_link_libraries(stage01_probe PRIVATE Eigen3::Eigen Ceres::ceres Sophus::Sophus)
# Exclude ROS/PCL utilities through their existing include guard. No upstream
# file is edited, copied, or mathematically replaced; see utility_bridge.hpp.
target_compile_options(stage01_probe PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/utility_bridge.hpp")
enable_testing()
add_test(NAME stage01_probe COMMAND stage01_probe)
