cmake_minimum_required(VERSION 3.24)
project(stage02_1_diagnostics LANGUAGES C CXX)
find_package(Eigen3 REQUIRED NO_MODULE)
find_package(Ceres 2.2 REQUIRED)
find_package(Sophus REQUIRED)
add_subdirectory(../stage02 stage02)
get_target_property(_inc stage02_reconstruct INCLUDE_DIRECTORIES)
get_target_property(_opts stage02_reconstruct COMPILE_OPTIONS)
add_executable(stage02_1_diagnostics diagnostics.cpp "${GPTR_ROOT}/gptr/include/SE3JQ.cpp")
target_compile_features(stage02_1_diagnostics PRIVATE cxx_std_17)
target_include_directories(stage02_1_diagnostics PRIVATE ${_inc} ../stage01 ../stage02)
target_compile_options(stage02_1_diagnostics PRIVATE ${_opts})
target_link_libraries(stage02_1_diagnostics PRIVATE Eigen3::Eigen Ceres::ceres Sophus::Sophus)
