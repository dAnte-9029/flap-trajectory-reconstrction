cmake_minimum_required(VERSION 3.24)
project(stage02_real_gptr LANGUAGES C CXX)
set(STAGE01_FIX_PSI_CUTOFF ON CACHE BOOL "Required audited PSI repair" FORCE)
find_package(Eigen3 REQUIRED NO_MODULE)
find_package(Ceres 2.2 REQUIRED)
find_package(Sophus REQUIRED)
add_subdirectory(../stage01 stage01)
get_target_property(_inc stage01_probe INCLUDE_DIRECTORIES)
get_target_property(_opts stage01_probe COMPILE_OPTIONS)
add_executable(stage02_reconstruct reconstruct.cpp "${GPTR_ROOT}/gptr/include/SE3JQ.cpp")
target_compile_features(stage02_reconstruct PRIVATE cxx_std_17)
target_include_directories(stage02_reconstruct PRIVATE ${_inc} ../stage01)
target_compile_options(stage02_reconstruct PRIVATE ${_opts})
target_link_libraries(stage02_reconstruct PRIVATE Eigen3::Eigen Ceres::ceres Sophus::Sophus)
# Imported targets found in a child directory are not always globally visible.
add_executable(stage02_verify verify.cpp "${GPTR_ROOT}/gptr/include/SE3JQ.cpp")
target_compile_features(stage02_verify PRIVATE cxx_std_17)
target_include_directories(stage02_verify PRIVATE ${_inc} ../stage01)
target_compile_options(stage02_verify PRIVATE ${_opts})
target_link_libraries(stage02_verify PRIVATE Eigen3::Eigen Ceres::ceres Sophus::Sophus)
enable_testing()
add_test(NAME stage02_pvt_factor COMMAND stage02_reconstruct --test)
